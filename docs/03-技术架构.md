## 26. 核心模块设计

### 26.1 FeishuClient（SDK 封装，单例）

```typescript
// src/lib/client.ts
import { Client } from '@larksuiteoapi/node-sdk'

let _client: Client | null = null

export function getFeishuClient(config: FeishuConfig): Client {
  if (!_client) {
    _client = new Client({
      appId: config.appId,
      appSecret: config.appSecret,
      loggerLevel: LoggerLevel.warn,  // 不打详细日志，防 App Secret 泄露
    })
  }
  return _client
}

export function resetClient() {
  _client = null
}
```

### 26.2 Token 管理

SDK 自动处理 token 刷新，应用只需：
- 构造 `Client` 时注入 `appId` + `appSecret`
- SDK 内部缓存 `tenant_access_token`（有效期 2 小时）
- 过期前 SDK 自动刷新

### 26.3 错误处理统一封装

```typescript
// src/lib/errors.ts

export class FeishuError extends Error {
  constructor(
    message: string,
    public code: string,
    public statusCode?: number
  ) {
    super(message)
    this.name = 'FeishuError'
  }
}

export function wrapFeishuError(e: any, defaultMsg: string): FeishuError {
  if (e instanceof FeishuError) return e
  if (e.code === 99991401) return new FeishuError('App Secret 错误', e.code)
  if (e.code === 99991664) return new FeishuError('权限不足', e.code)
  if (e.code === 230013) return new FeishuError('文件超限 20MB', e.code)
  return new FeishuError(defaultMsg, e.code || 'UNKNOWN')
}
```

## 27. Host Tools 注册

```typescript
// src/index.ts
import { installSettingsSection, settingsNamespace } from '@deepseek-ai/dsh-settings'
import { Schema } from '@deepseek-ai/schemastery'
import type { Context } from '@deepseek-ai/cordis'
import { feishuTools } from './tools/drive'

export const FEISHU_NS = settingsNamespace('dsh-feishu')

export interface Config {
  appId: string
  appSecret: string
  driveFolderToken?: string
  docxFolderToken?: string
  bitableFolderToken?: string
  uploadTimeout?: number
  maxFileSize?: number
  locale?: 'zh-CN' | 'en'
}

export const Config: Schema<Config> = Schema.object({
  appId:     Schema.string(),
  appSecret: Schema.string().role('secret'),
  driveFolderToken:  Schema.string().optional(),
  docxFolderToken:  Schema.string().optional(),
  bitableFolderToken: Schema.string().optional(),
  uploadTimeout: Schema.number().min(5000).max(300000).default(60000),
  maxFileSize:  Schema.number().min(1).max(20).default(20),
  locale:      Schema.enum(['zh-CN', 'en']).default('zh-CN'),
})

export function apply(ctx: Context, config: Config) {
  // 注册 host tools
  ctx.tools.register(feishuTools(config), { namespace: 'feishu' })

  // 注册 settings card
  installSettingsSection(ctx, FEISHU_NS, Config, config, {
    validate: async (value) => {
      // 验证 App ID 格式
      if (!value.appId.startsWith('cli_')) {
        throw new Error('App ID 必须以 cli_ 开头')
      }
    },
    setSource: (current) => { /* ... */ },
    onChange: () => { /* 连接变更时重置 client */ },
  })
}
```

## 28. Settings Card 注册（Client）

```typescript
// src/client/index.ts
import type { ClientContext } from '@deepseek-ai/dsh-client-runtime/client'

export const inject = ['slots', 'locale', 'connection', 'settingsScope']

export function apply(ctx: ClientContext): void {
  const card = new FeishuCardController(ctx.settingsScope.bind({
    namespace: 'dsh-feishu'
  }))

  ctx.slots.inject('settings.plugin.item', () =>
    ctx.slots.register({
      name: 'settings.plugin.item',
      key: 'dsh-feishu',
      locale: 'settings.dshFeishu',
      inject: () => card.inject(),
    }, FeishuCard)
  )
}
```

## 32. 安全审计清单

发布前必须确认：

- [ ] `appSecret` 字段有 `role('secret')` 标记
- [ ] `ctx.credentials.get()` 在运行时获取，不在任何日志/响应中出现
- [ ] SDK 日志级别设为 `warn`，不打印请求体/响应体
- [ ] `package.json` 无 `files: ['src']`（不发布源码，只发布构建产物）
- [ ] 无硬编码凭证（代码审计）
- [ ] `tsup.config.ts` 正确配置，不打包 `node_modules`
- [ ] README 包含安全说明，明确告知凭证不落盘
